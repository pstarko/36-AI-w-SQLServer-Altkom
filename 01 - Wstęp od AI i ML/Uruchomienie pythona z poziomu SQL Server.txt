    SaleID INT IDENTITY(1,1) PRIMARY KEY,
    ProductID INT NOT NULL,
    SaleDate DATE NOT NULL,
    Quantity INT NOT NULL,
    Revenue DECIMAL(10,2) NOT NULL
);

CREATE VIEW MonthlySales AS
SELECT 
    ProductID,
    YEAR(SaleDate) AS Year,
    MONTH(SaleDate) AS Month,
    SUM(Quantity) AS TotalQuantity,
    SUM(Revenue) AS TotalRevenue
FROM SalesHistory
GROUP BY ProductID, YEAR(SaleDate), MONTH(SaleDate);
go

-- Procedura składowana z modelem prognozowania
CREATE OR ALTER PROCEDURE PredictSales
    @ProductID INT,
    @ForecastMonths INT = 3
AS
BEGIN
    EXEC sp_execute_external_script
        @language = N'Python',
        @script = N'
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression

df = InputDataSet

if df.empty:
    # zawsze zwróć chociaż pustą ramkę danych o poprawnych kolumnach
    OutputDataSet = pd.DataFrame(columns=["ForecastMonth", "PredictedQuantity"])
else:
    # X = kolumna z indeksami miesięcy
    X = df[["MonthIndex"]].values
    y = df["TotalQuantity"].values

    model = LinearRegression().fit(X, y)

    # kolejne miesiące do prognozy
    last_idx = df["MonthIndex"].max()
    future_idx = np.arange(last_idx + 1, last_idx + 1 + forecast_months).reshape(-1, 1)

    predictions = model.predict(future_idx)

    result = pd.DataFrame({
        "ForecastMonth": np.arange(1, forecast_months + 1, dtype=int),
        "PredictedQuantity": predictions.astype(int)
    })
    OutputDataSet = result
',
        @input_data_1 = N'
            SELECT MonthIndex = ROW_NUMBER() OVER (ORDER BY Year, Month),
                   TotalQuantity
            FROM MonthlySales
            WHERE ProductID = @ProductID
            ORDER BY Year, Month
        ',
        @params = N'@ProductID INT, @forecast_months INT',
        @ProductID = @ProductID,
        @forecast_months = @ForecastMonths
    WITH RESULT SETS ((ForecastMonth INT, PredictedQuantity INT));
END;
GO
